cmake_minimum_required(VERSION 3.22)
project(PilonBreezeWorker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_library(audiocpp SHARED IMPORTED)
set_target_properties(audiocpp PROPERTIES IMPORTED_LOCATION "${AUDIOCPP_LIBRARY}")
add_executable(pilon-speech main.cpp)
target_include_directories(pilon-speech PRIVATE "${AUDIOCPP_INCLUDE_DIR}")
target_link_libraries(pilon-speech PRIVATE Qt6::Core audiocpp)
if(ANDROID)
    target_link_options(pilon-speech PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endif()
