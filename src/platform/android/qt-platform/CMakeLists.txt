cmake_minimum_required(VERSION 3.21)
project(PilonQtAndroidPlatform LANGUAGES C CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core CorePrivate Gui GuiPrivate)
if(NOT PILON_QT_BASE_SOURCE)
    message(FATAL_ERROR "Set PILON_QT_BASE_SOURCE to the verified, patched Qt source directory")
endif()
set(qpa "${PILON_QT_BASE_SOURCE}/src/plugins/platforms/android")
file(GLOB sources "${qpa}/*.cpp")
list(REMOVE_ITEM sources "${qpa}/extract-dummy.cpp")
add_library(PilonQtAndroidPlatform MODULE ${sources})
set_target_properties(PilonQtAndroidPlatform PROPERTIES OUTPUT_NAME plugins_platforms_qtforandroid_arm64-v8a)
target_include_directories(PilonQtAndroidPlatform PRIVATE "${qpa}" "${PILON_QT_BASE_SOURCE}/src/3rdparty/android")
target_compile_definitions(PilonQtAndroidPlatform PRIVATE QT_PLUGIN QT_NO_DEBUG)
target_link_libraries(PilonQtAndroidPlatform PRIVATE Qt6::Core Qt6::CorePrivate Qt6::Gui Qt6::GuiPrivate android jnigraphics EGL)
target_link_options(PilonQtAndroidPlatform PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
